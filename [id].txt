              {stageChecklist.length > 0 && (
                <div style={{
                  marginTop: '0.5rem',
                  padding: '12px 16px',
                  borderRadius: '12px',
                  backgroundColor: '#F8FAFC',
                  border: '1px solid #E2E8F0',
                  display: 'flex', flexDirection: 'column', gap: '6px'
                }}>
                  <div style={{ fontWeight: '700', color: '#0F172A', fontSize: '0.9rem', marginBottom: '4px' }}>
                    ✅ قائمة المتطلبات الإلزامية لهذه المرحلة
                  </div>
                  {stageChecklist.map((item, i) => (
                    <div key={i} style={{ display: 'flex', alignItems: 'center', gap: '10px', fontSize: '0.88rem' }}>
                      <span style={{
                        width: '20px', height: '20px', borderRadius: '50%',
                        border: `2px solid ${item.done ? '#10B981' : '#CBD5E1'}`,
                        backgroundColor: item.done ? '#10B981' : 'white',
                        color: 'white', display: 'flex', alignItems: 'center', justifyContent: 'center',
                        fontSize: '0.7rem', fontWeight: 'bold', transition: 'all 0.3s'
                      }}>
                        {item.done ? '✓' : i + 1}
                      </span>
                      <span style={{
                        color: item.done ? '#065F46' : '#475569',
                        textDecoration: item.done ? 'line-through' : 'none',
                        opacity: item.done ? 0.85 : 1
                      }}>
                        {item.text}
                      </span>
                    </div>
                  ))}
                </div>
              )}
            </div>

            {/* بطاقة المحادثة */}
            <div style={styles.chatCard} className="chat-card-layout">
              <div style={styles.chatCardHeader}>
                <div style={styles.clientStatus}>
                  <span style={styles.onlineDot}></span>
                  الدردشة مع: <strong>{scenario.clientName}</strong> · المرحلة {currentStage}/5
                </div>
                <div style={{ display: 'flex', gap: '8px', alignItems: 'center' }}>
                  {conversationScore.interactions > 0 && (
                    <div style={{
                      padding: '4px 12px', borderRadius: '14px',
                      backgroundColor: '#DBEAFE', color: '#1E40AF',
                      fontSize: '0.8rem', fontWeight: '700'
                    }}>
                      المتوسط: {conversationScore.average.toFixed(1)} / 10
                    </div>
                  )}
                  <div style={{ ...styles.difficultyBadge, ...getDifficultyBadgeStyle(scenario.difficulty) }}>
                    الصعوبة: {scenario.difficulty}
                  </div>
                </div>
              </div>

              {/* منطقة الرسائل */}
              <div style={styles.messagesArea}>
                <div style={styles.simulationStartPill}>
                  {scenario.projectLabel} · المحاكي يعمل حالياً بالذكاء الاصطناعي
                </div>

                {messages.map((msg) => {
                  const isUser = msg.sender === 'user';
                  const fb = msg.feedback;
                  const fbBadge = fb ? getFeedbackBadge(fb.type) : null;
                  return (
                    <div key={msg.id} style={isUser ? styles.userMessageRow : styles.clientMessageRow}>
                      {isUser ? (
                        <div style={styles.messageBubbleUser}>
                          {msg.text}
                          {msg.file && <div style={styles.fileAttachedPill}>📎 {msg.file}</div>}
                        </div>
                      ) : (
                        <>
                          <img src={scenario.avatar} alt={scenario.clientName} style={styles.avatar} />
                          <div style={{ display: 'flex', flexDirection: 'column', gap: '6px' }}>
                            <div style={styles.messageBubbleClient}>
                              <div style={{ whiteSpace: 'pre-line', lineHeight: '1.7' }}>
                                {msg.text}
                              </div>
                              {msg.file && <div style={styles.fileAttachedPill}>📎 {msg.file}</div>}
                            </div>
                            {fbBadge && (
                              <div style={{ display: 'flex', gap: '8px', alignItems: 'center', flexWrap: 'wrap' }}>
                                <span style={{
                                  padding: '2px 10px', borderRadius: '12px', fontSize: '0.75rem',
                                  fontWeight: '700', backgroundColor: fbBadge.bg, color: fbBadge.color
                                }}>
                                  {fbBadge.label} · {Math.round((msg.score || 0) * 100)}%
                                </span>
                                {fb?.note && (
                                  <span style={{
                                    fontSize: '0.78rem', color: '#64748B',
                                    padding: '2px 10px', backgroundColor: '#F1F5F9', borderRadius: '10px'
                                  }}>
                                    💬 {fb.note}
                                  </span>
                                )}
                              </div>
                            )}
                          </div>
                        </>
                      )}
                    </div>
                  );
                })}

                {/* منطقة رفع الملف للمرحلة 4 */}
                {currentStage === 4 && !uploadedFile && (
                  <div
                    onDragOver={onDragOver}
                    onDragLeave={onDragLeave}
                    onDrop={onDrop}
                    onClick={openFileDialog}
                    style={{
                      ...styles.dropZone,
                      borderColor: isDragging ? '#10B981' : '#CBD5E1',
                      backgroundColor: isDragging ? '#ECFDF5' : '#FFFFFF',
                      cursor: 'pointer'
                    }}
                  >
                    <input
                      type="file"
                      ref={fileInputRef}
                      style={{ display: 'none' }}
                      accept=".pkt,.pka"
                      onChange={onFilePicked}
                    />
                    <div style={{ fontSize: '56px', marginBottom: '8px' }}>📤</div>
                    <div style={{ fontSize: '1.2rem', fontWeight: '700', color: '#0F172A' }}>
                      منطقة رفع مشروع Packet Tracer
                    </div>
                    <div style={{ fontSize: '0.9rem', color: '#64748B', marginTop: '4px', textAlign: 'center', maxWidth: '420px' }}>
                      اسحب وأفلت الملف هنا <strong>أو</strong> اضغط في أي مكان داخل المربع لاختيار الملف
                    </div>
                    <div style={{
                      marginTop: '14px',
                      padding: '6px 14px',
                      backgroundColor: '#F0FDF4',
                      color: '#065F46',
                      borderRadius: '16px',
                      fontSize: '0.8rem',
                      fontWeight: '600',
                      border: '1px solid #86EFAC'
                    }}>
                      ✅ الملفات المسموحة: .pkt  ·  .pka (Packet Tracer)
                    </div>
                  </div>
                )}

                {isLoading && (
                  <div style={styles.clientMessageRow}>
                    <img src={scenario.avatar} alt={scenario.clientName} style={styles.avatar} />
                    <div style={styles.messageBubbleClient}>
                      <span style={styles.typingText}>يقوم {scenario.clientName} بالرد وتحليل رسالتك بواسطة الذكاء الاصطناعي...</span>
                    </div>
                  </div>
                )}
                <div ref={messagesEndRef} />
              </div>

              {/* منطقة الاقتراحات أسفل الرسائل */}
              {suggestions.length > 0 && (
                <div style={styles.suggestionsBar}>
                  <div style={{ fontSize: '0.8rem', fontWeight: '700', color: '#475569', marginBottom: '8px' }}>
                    💡 اقتراحات للمرحلة الحالية
                  </div>
                  <div style={{ display: 'flex', flexWrap: 'wrap', gap: '8px' }}>
                    {suggestions.map((s, i) => (
                      <button
                        key={i}
                        onClick={() => handleSend(s)}
                        disabled={isLoading}
                        style={styles.suggestionPill}
                      >
                        {s}
                      </button>
                    ))}
                  </div>
                </div>
              )}

              {/* منطقة إدخال النص */}
              <div style={styles.inputContainer}>

                {currentStage === 4 && uploadedFile && (
                  <div style={{
                    marginBottom: '12px', padding: '12px 14px',
                    borderRadius: '12px', backgroundColor: '#ECFDF5',
                    border: '1px solid #6EE7B7',
                    display: 'flex', justifyContent: 'space-between', alignItems: 'center', gap: '10px', flexWrap: 'wrap'
                  }}>
                    <div>
                      <div style={{ fontWeight: '700', color: '#065F46', fontSize: '0.92rem' }}>
                        📎 الملف المختار: {uploadedFile.name}
                      </div>
                      <div style={{ fontSize: '0.82rem', color: '#047857', marginTop: '2px' }}>
                        حجم الملف: {Math.round(uploadedFile.size / 1024)} كيلوبايت
                      </div>
                    </div>
                    <div style={{ display: 'flex', gap: '8px' }}>
                      <button
                        onClick={() => { setUploadedFile(null); setConfirmUploadMessage(''); }}
                        style={{
                          padding: '6px 14px', borderRadius: '8px',
                          border: '1px solid #D1D5DB', backgroundColor: 'white',
                          color: '#374151', fontWeight: '600', cursor: 'pointer'
                        }}
                      >
                        تغيير الملف
                      </button>
                      <button
                        onClick={sendUploadedFile}
                        disabled={isLoading}
                        style={{
                          padding: '6px 18px', borderRadius: '8px',
                          backgroundColor: '#10B981', color: 'white',
                          fontWeight: '700', border: 'none', cursor: isLoading ? 'not-allowed' : 'pointer',
                          opacity: isLoading ? 0.6 : 1
                        }}
                      >
                        إرسال الملف للعميل →
                      </button>
                    </div>
                  </div>
                )}

                {confirmUploadMessage && (
                  <div style={{
                    marginBottom: '10px', fontSize: '0.85rem', color: '#1F2937',
                    padding: '6px 12px', borderRadius: '8px', backgroundColor: '#FEF9C3',
                    border: '1px solid #FDE047'
                  }}>
                    {confirmUploadMessage}
                  </div>
                )}

                <textarea
                  value={inputValue}
                  onChange={(e) => setInputValue(e.target.value)}
                  placeholder={
                    currentStage === 1 ? 'اكتب سؤالك للعميل... (مثال: كم عدد الأجهزة المتوقعة؟)' :
                    currentStage === 2 ? 'اكتب عرض السعر والتبرير...' :
                    currentStage === 3 ? 'أجب على سؤال التسليم الحالي...' :
                    currentStage === 4 ? (uploadedFile ? 'اكتب تعليقاً إضافياً مع الملف...' : '💡 استخدم مربع رفع الملف بالأعلى لرفع مشروع Packet Tracer') :
                    'اكتب رسالة ختامية...'
                  }
                  style={styles.textArea}
                  onKeyDown={(e) => {
                    if (e.key === 'Enter' && !e.shiftKey) {
                      e.preventDefault();
                      handleSend();
                    }
                  }}
                  disabled={isLoading}
                />

                <div style={styles.inputActionBar}>
                  <div style={styles.rightActions}>
                    <button onClick={openFileDialog} style={styles.toolBtn}>
                      📎 إرفاق ملف Packet Tracer
                    </button>
                  </div>
                  <div style={styles.leftActions}>
                    <button onClick={() => setShowTips(!showTips)} style={styles.tipsBtn}>
                      💡 تلميحات
                    </button>
                    <button
                      onClick={() => handleSend()}
                      style={styles.sendBtn}
                      disabled={isLoading || (!inputValue.trim() && currentStage === 4 && !uploadedFile)}
                    >
                      {isLoading ? '...جاري المعالجة' : `إرسال إلى ${scenario.clientName} ◀`}
                    </button>
                  </div>
                </div>

                <div style={styles.evaluationFooterRow}>
                  <span style={styles.evaluationText}>
                    يتم تقييم:{' '}
                    <strong style={{ color: '#032639' }}>
                      {currentStage === 1 && 'شمولية أسئلتك للمتطلبات'}
                      {currentStage === 2 && 'منطق التسعير وحججك التفاوضية'}
                      {currentStage === 3 && 'وضوح إجاباتك عن أسئلة التسليم'}
                      {currentStage === 4 && 'جودة ملف التصميم المرفوع'}
                      {currentStage === 5 && 'إحترافية الختام والتعامل'}
                    </strong>
                  </span>
                  <p style={styles.projectMiniLabel}>{scenario.projectLabel}</p>
                </div>
              </div>
            </div>
          </section>

          {/* ====== الشريط الجانبي: ملف العميل ====== */}
          <aside style={styles.sidebarSection} className="sidebar-layout">
            <div style={styles.scenarioBriefCard}>
              <div style={styles.scrollContainer}>
                <h2 style={styles.briefTitle}>{scenario.title}</h2>

                <div style={styles.clientProfileArea}>
                  <img src={scenario.avatar} alt={scenario.clientName} style={styles.largeAvatar} />
                  <h3 style={styles.clientNameTxt}>{scenario.clientName}</h3>
                  <p style={styles.clientRoleTxt}>{scenario.role}</p>
                </div>

                <div style={styles.personalitySection}>
                  <div style={styles.personalityHeader}>السمات الشخصية</div>
                  <div style={styles.badgesWrapper}>
                    {scenario.personalityTags.map((tag, tIdx) => (
                      <span key={tIdx} style={{ ...styles.personalityBadgeItem, backgroundColor: tag.bg, color: tag.color }}>
                        {tag.text}
                      </span>
                    ))}
                  </div>
                </div>

                <div style={{
                  padding: '12px 14px', backgroundColor: '#EFF6FF',
                  border: '1px solid #BFDBFE', borderRadius: '12px',
                  marginBottom: '1rem', textAlign: 'center'
                }}>
                  <div style={{ fontSize: '0.9rem', fontWeight: '700', color: '#1E40AF' }}>
                    🎯 المرحلة الحالية ({currentStage}/5)
                  </div>
                  <div style={{ fontSize: '1.05rem', fontWeight: '800', color: '#0F172A', marginTop: '4px' }}>
                    {FIVE_STAGES[currentStage - 1]?.icon} {FIVE_STAGES[currentStage - 1]?.label}
                  </div>
                </div>

                <div style={styles.projectContextBox}>
                  <div style={styles.contextHeader}>
                    <svg width="16" height="16" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2.5" style={{ marginLeft: '6px' }}>
                      <path d="M14 2H6a2 2 0 0 0-2 2v16a2 2 0 0 0 2 2h12a2 2 0 0 0 2-2V8z"></path>
                      <polyline points="14 2 14 8 20 8"></polyline>
                    </svg>
                    سياق المشروع
                  </div>
                  <p style={styles.contextText}>{scenario.description} {scenario.personality}</p>
                </div>

                <div style={styles.divider}></div>

                <div style={styles.featuresSection}>
                  <div style={styles.featuresHeader}>خصائص السيناريو</div>
                  <div style={styles.featuresGrid}>
                    {scenario.features.map((feat, idx) => (
                      <div key={idx} style={feat.highlight ? styles.featurePillHighlight : styles.featurePillNormal}>
                        <span style={{ marginLeft: '6px' }}>{feat.icon}</span>
                        {feat.text}
                      </div>
                    ))}
                  </div>
                </div>

                <div style={styles.divider}></div>

                <div style={styles.tasksSection}>
                  <h3 style={styles.tasksHeader}>📋 المطلوب العام:</h3>
                  <ul style={styles.tasksList}>
                    {scenario.tasks.map((task, index) => (
                      <li key={index} style={styles.taskItem}>
                        <span style={styles.taskDot}>•</span>
                        <span style={styles.taskText}>{task}</span>
                      </li>
                    ))}
                  </ul>
                </div>

                <div style={{
                  marginTop: '16px', padding: '12px 16px',
                  backgroundColor: '#F0F7F8', borderRadius: 12,
                  border: '1px solid #17919e'
                }}>
                  <div style={{ fontSize: 13, fontWeight: 700, color: '#0D1E3B' }}>
                    🎯 مستواك الحالي:
                    <span style={{
                      backgroundColor: userLevel === 'advanced' ? '#2ECC71' : userLevel === 'intermediate' ? '#F39C12' : '#17919e',
                      color: 'white', padding: '2px 12px', borderRadius: 12,
                      marginRight: 8, fontSize: 12
                    }}>
                      {userLevel === 'advanced' ? 'متقدم' : userLevel === 'intermediate' ? 'متوسط' : 'مبتدئ'}
                    </span>
                  </div>
                  <div style={{ fontSize: 12, color: '#64748B', marginTop: 4 }}>
                    {userLevel === 'advanced' && '🌟 مستوى عالٍ. استخدم حججاً تقنية متقدمة.'}
                    {userLevel === 'intermediate' && '📈 مستوى جيد. استمر في التحدي.'}
                    {userLevel === 'beginner' && '🌱 بداية رائعة! اسأل الأسئلة الخمس في المرحلة الأولى.'}
                  </div>
                </div>
              </div>

              <div style={styles.sidebarFooter}>
                التركيز في هذه المرحلة: {scenario.focus}
              </div>
            </div>
          </aside>
        </div>
      </main>

      {/* نافذة التلميحات */}
      {showTips && (
        <div style={styles.modalOverlay} onClick={() => setShowTips(false)}>
          <div style={styles.interactiveTipsBox} onClick={e => e.stopPropagation()}>
            <div style={styles.tipsHeaderRow}>
              <div style={styles.tipsTitle}><span>💡 تلميحات للمرحلة {currentStage} · {FIVE_STAGES[currentStage - 1]?.label}</span></div>
              <button onClick={() => setShowTips(false)} style={styles.closeTipsBtn}>✕</button>
            </div>
            <div style={styles.tipsContent}>
              {currentStage === 1 && (
                <>
                  <div style={styles.tipTextRow}><span style={styles.tipNumber}>1.</span><p style={styles.tipText}>اسأل الأمانة الخمس الإلزامية: الأجهزة، الإنترنت، شبكة الضيوف، الميزانية، المساحة.</p></div>
                  <div style={styles.tipTextRow}><span style={styles.tipNumber}>2.</span><p style={styles.tipText}>لا تقدم عرض سعر قبل ما تكتشف كل متطلبات العميل — هذا خطأ مهني شائع.</p></div>
                  <div style={styles.tipTextRow}><span style={styles.tipNumber}>3.</span><p style={styles.tipText}>بعد كل سؤال، اسجل الإجابة عقلياً وانتقل للسؤال التالي حتى تكتمل الخمسة.</p></div>
                  <div style={styles.tipTextRow}><span style={styles.tipNumber}>4.</span><p style={styles.tipText}>إذا كنت متأكداً من إجاباتك على الخمسة، سيقودك الذكاء الاصطناعي آلياً للمرحلة الثانية.</p></div>
                </>
              )}
              {currentStage === 2 && (
                <>
                  <div style={styles.tipTextRow}><span style={styles.tipNumber}>1.</span><p style={styles.tipText}>لا تعطِ السعر رقمًا فقط — ابدأ بتقديم قائمة التجهيزات ثم السعر الإجمالي.</p></div>
                  <div style={styles.tipTextRow}><span style={styles.tipNumber}>2.</span><p style={styles.tipText}>أبرز جودة الأجهزة والضمان والدعم الفني كحجج لتبرير السعر.</p></div>
                  <div style={styles.tipTextRow}><span style={styles.tipNumber}>3.</span><p style={styles.tipText}>إذا طلب العميل خصماً، لا ترفض مباشرة — اربط الخصم بشرط (مثل الدفع المسبق).</p></div>
                </>
              )}
              {currentStage === 3 && (
                <>
                  <div style={styles.tipTextRow}><span style={styles.tipNumber}>1.</span><p style={styles.tipText}>أجب عن أسئلة التسليم الثلاثة بوضوح وتفاصيل عملية (موعد، مدة ضمان، نوع التعديلات).</p></div>
                  <div style={styles.tipTextRow}><span style={styles.tipNumber}>2.</span><p style={styles.tipText}>مثال: "التسليم خلال أسبوعين، شهر تعديلات مجانية، وتعديلات البرمجة والضبط ضمن الضمان."</p></div>
                </>
              )}
              {currentStage === 4 && (
                <>
                  <div style={styles.tipTextRow}><span style={styles.tipNumber}>1.</span><p style={styles.tipText}>استخدم منطقة رفع الملف في أعلى منطقة المحادثة.</p></div>
                  <div style={styles.tipTextRow}><span style={styles.tipNumber}>2.</span><p style={styles.tipText}>تأكد أن جميع الأجهزة متصلة، وأن عناوين IP مدخلة بشكل صحيح قبل الرفع.</p></div>
                  <div style={styles.tipTextRow}><span style={styles.tipNumber}>3.</span><p style={styles.tipText}>بعد الرفع، سيقول العميل "تم الاستلام" وينتقل معك للمرحلة النهائية.</p></div>
                </>
              )}
              {currentStage === 5 && (
                <>
                  <div style={styles.tipTextRow}><span style={styles.tipNumber}>1.</span><p style={styles.tipText}>أغلق الصفقة بإيجابية، واشكر العميل، وأذكر أنك متاح للدعم المستقبلي.</p></div>
                  <div style={styles.tipTextRow}><span style={styles.tipNumber}>2.</span><p style={styles.tipText}>يمكنك طلب مراجعة من العميل — هذه ممارسة مهنية جيدة تُظهر الثقة.</p></div>
                </>
              )}
            </div>
            <div style={styles.tipsFooter}>
              <button onClick={() => setShowTips(false)} style={styles.understandBtn}>حسناً، فهمت</button>
            </div>
          </div>
        </div>
      )}

      {/* التذييل */}
      <footer style={styles.footer}>
        <div style={styles.footerContainer}>
          <div style={styles.footerLeft}>
            <p style={styles.footerBrand}>SmartLab</p>
            <p style={styles.footerText}>منصة تعليمية متطورة — محاكي عميل ذكي مدعوم بالذكاء الاصطناعي عبر 5 مراحل متتالية.</p>
          </div>
        </div>
      </footer>
    </div>
  );
}
