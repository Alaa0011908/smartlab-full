// Fully AI-powered conversation endpoint with 5-stage workflow
// Uses ScenarioEngine which calls Deepseek LLM internally
// ============================================================

import { ScenarioEngine } from '../../lib/scenarioEngine';

// Max history to keep for context
const MAX_HISTORY = 16;

function trimHistory(history) {
    if (!Array.isArray(history)) return [];
    return history.slice(-MAX_HISTORY);
}

export default async function handler(req, res) {
    // CORS headers
    res.setHeader('Access-Control-Allow-Origin', '*');
    res.setHeader('Access-Control-Allow-Methods', 'POST, OPTIONS');
    res.setHeader('Access-Control-Allow-Headers', 'Content-Type');

    // Handle preflight
    if (req.method === 'OPTIONS') {
        return res.status(200).end();
    }

    // Only POST allowed
    if (req.method !== 'POST') {
        return res.status(405).json({ 
            error: 'Method not allowed', 
            message: 'استخدم POST' 
        });
    }

    try {
        const {
            scenarioId = 'cafe',
            message = '',
            history = [],
            userLevel = 'beginner',
            conversationStage = 1,
            stageDataSnapshot = null,
            fileInfo = null,
            sessionId = null
        } = req.body || {};

        // ---- Validation ----
        if (!scenarioId || typeof scenarioId !== 'string') {
            return res.status(400).json({ 
                error: 'Invalid scenarioId',
                message: 'معرف السيناريو غير صحيح'
            });
        }
        
        if (typeof message !== 'string') {
            return res.status(400).json({ 
                error: 'Invalid message',
                message: 'الرسالة غير صحيحة'
            });
        }

        // ---- Log API call ----
        console.log('[API] Scenario:', scenarioId, 'Stage:', conversationStage, 'Message length:', message.length);
        console.log('[API] File info:', fileInfo);
        console.log('[API] DeepSeek Key present:', !!process.env.DEEPSEEK_API_KEY);

        // ---- Hydrate stage state from client ----
        const hydratedStageData = ScenarioEngine.hydrateStageData(stageDataSnapshot);
        const trimmedHistory = trimHistory(history);

        // ---- File upload detection ----
        const fileUploaded = Boolean(fileInfo && fileInfo.uploaded);
        const fileName = fileInfo && fileInfo.name ? fileInfo.name : null;

        // ---- Instantiate engine and run ----
        const engine = new ScenarioEngine(
            scenarioId,
            trimmedHistory,
            userLevel,
            conversationStage,
            hydratedStageData
        );

        const response = await engine.analyzeAndRespond(message, { fileUploaded, fileName });

        // ---- Build response ----
        return res.status(200).json({
            reply: response.text,
            message: response.text,
            feedback: response.feedback,
            score: response.score,
            suggestions: response.suggestions || [],
            stage: response.stage,
            stageName: response.stageName,
            stageChanged: response.stageChanged,
            previousStage: response.previousStage,
            stageDataSnapshot: response.stageDataSnapshot,
            meta: {
                aiProvider: response.meta?.provider,
                usedFallback: response.meta?.usedFallback,
                aiMeta: response.meta?.aiMeta || {}
            },
            timestamp: new Date().toISOString(),
            ok: true
        });

    } catch (error) {
        console.error('[scenario.js] API Error:', error);
        return res.status(500).json({
            error: 'Server error',
            message: 'عذراً، حدث خطأ في معالجة المحادثة. حاول مرة أخرى بعد لحظات.',
            ok: false,
            details: process.env.NODE_ENV === 'development' ? error.message : undefined
        });
    }
}

// Disable default body parser size limits
export const config = {
    api: {
        bodyParser: {
            sizeLimit: '4mb'
        }
    }
};
